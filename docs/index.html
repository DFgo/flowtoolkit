FlowToolkit 文档

OA系统前端工具库 - 提供水印、多语言、工具函数等常用功能

版本: 2.0.0

Watermark 水印模块

导入方式

import { Watermark } from 'flowtoolkit';

创建水印

const wm = new Watermark({
  content: 'OA系统 机密文件',
  color: 'rgba(0, 0, 0, 0.3)',
  fontSize: 14,
  angle: -30,
  scale: 2,
  protect: true,
  password: '123456',
  onSuccess: () => console.log('水印创建成功'),
  onCancel: () => console.log('用户取消')
});

// 创建水印
wm.create();

方法

wm.create() - 创建水印
wm.update({ content: '新内容' }) - 更新水印内容
wm.destroy() - 销毁水印
wm.recover() - 恢复水印
wm.getOptions() - 获取当前配置
wm.isCreated() - 检查是否已创建
wm.isVerified() - 检查是否已验证密码

i18n 国际化模块

导入方式

import { i18n } from 'flowtoolkit';

初始化

i18n.init({
  defaultLang: 'zh-CN',
  currentLang: 'zh-CN',
  customMessages: {
    'en': { 'hello': 'Hello', 'world': 'World' },
    'zh-CN': { 'hello': '你好', 'world': '世界' }
  }
});

方法

i18n.t('key') - 获取翻译 参数: key, variables?
i18n.changeLang('en') - 切换语言 参数: lang
i18n.refresh() - 刷新页面翻译
i18n.replaceElement(target) - 替换单个元素 参数: selector | element
i18n.extendMessages('zh-CN', { 'key': 'value' }) - 扩展语言包
i18n.getCurrentLang() - 获取当前语言
i18n.getLoadedLangs() - 获取已加载语言列表
i18n.reset() - 重置配置

HTML 使用

<!-- 文本替换 -->
<span data-i18n="hello">你好</span>

<!-- 指定替换属性 -->
<input data-i18n="placeholder" data-i18n-prop="placeholder">

<!-- 带变量 -->
<span data-i18n="welcome" data-i18n-name="张三"></span>

Utils 工具模块

导入方式

// 导入整个 utils 对象
import { utils } from 'flowtoolkit';

// 或直接导入具体函数
import { $, $$, format, addDays } from 'flowtoolkit';

DOM 工具 - utils.dom

utils.dom.$('#id') - querySelector
utils.dom.$$('.class') - querySelectorAll
utils.dom.create('div') - 创建元素
utils.dom.addClass(el, 'active') - 添加类名
utils.dom.removeClass(el, 'active') - 移除类名
utils.dom.toggleClass(el, 'active') - 切换类名
utils.dom.hasClass(el, 'active') - 检查类名
utils.dom.on(el, 'click', fn) - 绑定事件
utils.dom.off(el, 'click', fn) - 解绑事件
utils.dom.one(el, 'click', fn) - 绑定一次性事件

Date 工具 - utils.date

utils.date.format(date, 'YYYY-MM-DD') - 格式化日期
utils.date.parse('2024-01-01') - 解析日期
utils.date.addDays(date, 5) - 加天数
utils.date.addMonths(date, 1) - 加月数
utils.date.addYears(date, 1) - 加年数
utils.date.diffDays(date1, date2) - 天数差
utils.date.diffMonths(date1, date2) - 月数差
utils.date.relative(date) - 相对时间 (如: 2小时前)
